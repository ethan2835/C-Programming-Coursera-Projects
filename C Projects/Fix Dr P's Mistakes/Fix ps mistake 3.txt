#include <stdio.h>
#include <math.h> // Correctly included for sin() and fabs()

int main(void)
{
    double interval;
    int i;

    for(i = 0; i < 30; i++)
    {
        // Use 10.0 to ensure floating-point division
        interval = i / 10.0; 
        
        // Use fabs() instead of abs() for double types
        printf("sin( %.2lf ) = %.3lf \n", interval, fabs(sin(interval)));
    }

    printf("\n+++++++\n");
    return 0;
}
